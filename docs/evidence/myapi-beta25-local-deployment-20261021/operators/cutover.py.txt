def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1048576), b''):
            h.update(block)
    return h.hexdigest()

def local_get(path):
    require(path.startswith('/') and not path.startswith('//'), 'NONLOCAL_PROBE_PATH')
    request = urllib.request.Request('http://127.0.0.1:3000' + path, headers={'Accept-Encoding': 'identity'})
    with opener.open(request, timeout=4) as response:
        require(response.geturl().startswith('http://127.0.0.1:3000/'), 'LOCAL_PROBE_REDIRECTED')
        body = response.read(20000001)
        require(len(body) <= 20000000, 'PROBE_RESPONSE_TOO_LARGE')
        return response.status, response.headers.get('Content-Type', ''), body

def native_backup(source, target):
    src = sqlite3.connect(source.as_uri() + '?mode=ro', uri=True, timeout=10)
    src.execute('PRAGMA query_only=ON')
    dst = sqlite3.connect(target)
    started = time.monotonic()
    def progress(status, remaining, total):
        require(time.monotonic() - started < 45, 'NATIVE_BACKUP_TIMEOUT')
    src.backup(dst, pages=256, progress=progress, sleep=0.05)
    require(dst.execute('PRAGMA integrity_check').fetchall() == [('ok',)], 'DATABASE_INTEGRITY_FAILED')
    require(not dst.execute('PRAGMA foreign_key_check').fetchall(), 'DATABASE_FOREIGN_KEYS_FAILED')
    dst.close()
    src.close()
    target.chmod(0o600)

def stop_without_kill(name, maximum):
    # --time=-1 records a manual stop and never escalates to SIGKILL.
    p = subprocess.Popen(DOCKER + ['stop', '--time', '-1', name], env=docker_env,
                         stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    try:
        p.communicate(timeout=maximum)
    except subprocess.TimeoutExpired:
        p.terminate()
        p.communicate(timeout=5)
        raise RuntimeError('GRACEFUL_DRAIN_NOT_COMPLETED_NO_FORCE_KILL')
    require(p.returncode == 0, 'GRACEFUL_STOP_FAILED')
    result = inspect(name)
    require(not result['State']['Running'] and result['State']['ExitCode'] == 0 and not result['State']['OOMKilled'], 'CONTAINER_DID_NOT_EXIT_CLEANLY')
    return result

            rawlogs = subprocess.run(DOCKER + ['logs', NAME], env=docker_env, capture_output=True, timeout=15)
            (backup / 'failed-new-container-console.private.log').write_bytes(rawlogs.stdout + rawlogs.stderr)
            report['failed_new_service_stopped_scene_preserved'] = True
            report['rollback'] = 'Not automatic after new service execution; credentials/business/outbound state must be reviewed.'
        except Exception:
            report['failed_new_service_stop_requires_attention'] = True
    elif old_stopped:
        docker(['start', TARGET])
        report['rollback'] = 'Original service restarted; candidate process never started and original database was not modified.'
    report['new_container_created'] = new_created
finally:
    report['finished_at'] = datetime.now(timezone.utc).isoformat()
    if report.get('first_beta25_http_200_at') and report.get('old_stop_requested_at'):
        report['http_interruption_upper_bound_seconds'] = round((datetime.fromisoformat(report['first_beta25_http_200_at']) - datetime.fromisoformat(report['old_stop_requested_at'])).total_seconds(), 3)
    OUT.write_text(json.dumps(report, indent=2) + '\n')
    print(json.dumps({'success': report['success'], 'deployed': report['deployed'], 'failure_code': report.get('failure_code'),
                      'new_container_id': report.get('new_container_id'), 'version': report.get('public_status', {}).get('version'),
                      'http_interruption_upper_bound_seconds': report.get('http_interruption_upper_bound_seconds'),
                      'result': str(OUT), 'final_backup_directory': str(backup), 'new_runtime_directory': str(runtime)}), flush=True)
    raise SystemExit(0 if report['success'] else 1)
