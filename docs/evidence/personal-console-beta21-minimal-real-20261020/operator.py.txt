def require(value, code):
    if not value:
        raise RuntimeError(code)

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None

def request(method, path, body=None, credential=None, timeout=20):
    require(path.startswith('/') and not path.startswith('//'), 'LOCAL_PATH_INVALID')
    headers = {'Origin': origin, 'Content-Type': 'application/json', 'Accept-Encoding': 'identity'}
    credential = access if credential is None else credential
    if credential:
        headers['Authorization'] = 'Bearer '+credential
    if credential == access and sid:
        headers['X-Auth-Session'] = sid
    data = None if body is None else json.dumps(body).encode()
    req = urllib.request.Request(origin+path, method=method, headers=headers, data=data)
    result['local_requests'] += 1
    try:
        response = local_opener.open(req, timeout=timeout)
    except urllib.error.HTTPError as e:
        response = e
    with response:
        raw = response.read(2*1024*1024+1)
        require(len(raw) <= 2*1024*1024, 'LOCAL_RESPONSE_TOO_LARGE')
        try: payload = json.loads(raw)
        except (ValueError, UnicodeDecodeError): raise RuntimeError('LOCAL_RESPONSE_NOT_JSON')
        return response.status, payload

def api(method, path, body=None):
    status, payload = request(method, path, body)
    if status != 200 or payload.get('success') is not True:
        result['local_refusal'] = {'http_status': status, 'phase': result['phase']}
        raise RuntimeError('LOCAL_API_REFUSED')
    return payload.get('data')

def counters(value):
    r = value['run']
    return {**{n:r[n] for n in ['status','total','auth','refresh','sample','models','generation','generation_in_flight']},
            **{n:value[n] for n in ['owner_current','operations','http_in_flight','drain_error']}}

def checkpoint(total, samples, generations):
    value = status()
    snapshot = counters(value)
    result['run_counters'] = snapshot
    save()
    require(value['run']['total'] == total and value['run']['sample'] == samples and value['run']['generation'] == generations,
            'RUN_COUNTER_MISMATCH')
    require(all(value['run'][n] == 0 for n in ['auth','refresh','models','generation_in_flight']) and
            value['operations'] == 0 and value['http_in_flight'] == 0 and not value['drain_error'], 'RUN_NOT_IDLE')
    print(json.dumps({'run_counters': snapshot}), flush=True)

    result['samples'].append(sample)

            'USAGE_QUERY_FAILED_STOP_NO_RETRY')
    data = payload.get('data')
    require(isinstance(data, dict), 'USAGE_PAYLOAD_INVALID')
    sample['plan_type'] = data.get('plan_type') if data.get('plan_type') in ['free','plus','pro','team','business','enterprise','edu'] else 'other'
    windows = data.get('rate_limit') or {}

def generate():
    phase('single_generation')
    body = {'model':'gpt-6-luna', 'instructions':'', 'input':[{'role':'user','content':[{'type':'input_text','text':'只回复 OK'}]}],
            'stream':True, 'store':False}
    req = urllib.request.Request(origin+'/v1/responses', method='POST', data=json.dumps(body).encode(),
            headers={'Authorization':'Bearer '+key,'Content-Type':'application/json','Accept':'text/event-stream','Accept-Encoding':'identity'})
    result['local_requests'] += 1
    try: response = local_opener.open(req, timeout=100)
    except urllib.error.HTTPError as e: response = e
    generation = {'local_http':response.status, 'response_completed':False, 'usage_reported':False, 'output_exact_ok':False}
    result['generation'] = generation
    with response:
        if response.status != 200:
            raw = response.read(65537)
            try:
                err = json.loads(raw).get('error') or {}
                safe_codes = {'insufficient_quota','model_not_found','invalid_api_key','invalid_request_error','upstream_acceptance_denied','usage_dispatch_unresolved'}
                generation['error_code'] = err.get('code') if err.get('code') in safe_codes else 'unspecified'
            except (ValueError, AttributeError): generation['error_code'] = 'unspecified'
            raise RuntimeError('GENERATION_REFUSED_STOP_NO_RETRY')
        text = []
        byte_count = 0
        for line in response:
            byte_count += len(line)
            require(byte_count <= 2*1024*1024, 'GENERATION_STREAM_TOO_LARGE')
            if not line.startswith(b'data:'): continue
            item = line[5:].strip()
            if item == b'[DONE]': continue
            try: event = json.loads(item)
            except ValueError: raise RuntimeError('GENERATION_STREAM_INVALID')
            kind = event.get('type')
            if kind == 'response.output_text.delta' and isinstance(event.get('delta'),str): text.append(event['delta'])
            if kind in ['error','response.failed','response.incomplete']:
                raise RuntimeError('GENERATION_TERMINAL_FAILURE_STOP_NO_RETRY')
            if kind == 'response.completed':
                completed = event.get('response') or {}
                generation['response_completed'] = completed.get('status') == 'completed'
                usage_data = completed.get('usage')
                if isinstance(usage_data,dict):
                    generation['usage_reported'] = True
                    for name in ['input_tokens','output_tokens','total_tokens']:
                        if type(usage_data.get(name)) is int and 0 <= usage_data[name] <= 2147483647:
                            generation[name] = usage_data[name]
        generation['stream_bytes'] = byte_count
        generation['output_exact_ok'] = ''.join(text).strip() == 'OK'
        text.clear()
        require(generation['response_completed'], 'GENERATION_COMPLETION_NOT_PROVEN')
        require(generation['output_exact_ok'], 'GENERATION_OUTPUT_UNEXPECTED_STOP')
    save()
    print(json.dumps({'generation': generation}), flush=True)
