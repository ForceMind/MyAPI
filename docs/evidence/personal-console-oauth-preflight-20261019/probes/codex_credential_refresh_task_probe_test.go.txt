func TestProbeAutoRefreshTreatsValidOneHourCredentialAsDue(t *testing.T) {
	previousDB := model.DB
	previousMemoryCacheEnabled := common.MemoryCacheEnabled
	db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	require.NoError(t, err)
	require.NoError(t, db.AutoMigrate(&model.Channel{}, &model.SystemTaskLock{}))
	model.DB = db
	common.MemoryCacheEnabled = false
	codexCredentialRefreshRunning.Store(false)
	t.Cleanup(func() {
		model.DB = previousDB
		common.MemoryCacheEnabled = previousMemoryCacheEnabled
		codexCredentialRefreshRunning.Store(false)
	})

	channel := &model.Channel{
		Type:   constant.ChannelTypeCodex,
		Name:   "one-hour-refresh-probe",
		Status: common.ChannelStatusEnabled,
		Key: `{"access_token":"old-access","refresh_token":"old-refresh","account_id":"account-123456","expired":"` +
			time.Now().Add(25*time.Hour).Format(time.RFC3339) + `"}`,
	}
	require.NoError(t, db.Create(channel).Error)
	establishCodexCredentialSubjectEvidence(t, db, channel.Id, channel.Key, "account-123456")

	client, err := GetHttpClientWithProxy("")
	require.NoError(t, err)
	previousTransport := client.Transport
	t.Cleanup(func() { client.Transport = previousTransport })
	var tokenRequests atomic.Int32
	client.Transport = codexCredentialRefreshRoundTripper(func(request *http.Request) (*http.Response, error) {
		require.Equal(t, "/oauth/token", request.URL.Path)
		requestNumber := tokenRequests.Add(1)
		accessToken := codexCredentialTestJWT(t, "account-123456")
		return &http.Response{
			StatusCode: http.StatusOK,
			Header:     make(http.Header),
			Body:       io.NopCloser(strings.NewReader(`{"access_token":"` + accessToken + `","refresh_token":"new-refresh-` + strconv.Itoa(int(requestNumber)) + `","expires_in":3600}`)),
			Request:    request,
		}, nil
	})

	runCodexCredentialAutoRefreshOnce()
	require.Zero(t, tokenRequests.Load(), "25h credential must not be refreshed")

	key, err := parseCodexOAuthKey(channel.Key)
	require.NoError(t, err)
	key.Expired = time.Now().Add(time.Hour).Format(time.RFC3339)
	encoded, err := common.Marshal(key)
	require.NoError(t, err)
	channel.Key = string(encoded)
	require.NoError(t, db.Model(channel).Update("key", channel.Key).Error)
	establishCodexCredentialSubjectEvidence(t, db, channel.Id, channel.Key, "account-123456")

	runCodexCredentialAutoRefreshOnce()
	require.Equal(t, int32(1), tokenRequests.Load())
	runCodexCredentialAutoRefreshOnce()
	require.Equal(t, int32(2), tokenRequests.Load(), "refreshed 1h token remains within the 24h threshold on another pass")
	t.Logf("25h credential dispatched=0; 1h credential dispatched=2 across two explicit passes (no ticker or sleep)")
	var stored model.Channel
	require.NoError(t, db.First(&stored, channel.Id).Error)
	storedKey, err := parseCodexOAuthKey(stored.Key)
	require.NoError(t, err)
	require.Equal(t, "new-refresh-2", storedKey.RefreshToken)
	require.WithinDuration(t, time.Now().Add(time.Hour), mustParseCodexProbeTime(t, storedKey.Expired), 5*time.Second)
}

func mustParseCodexProbeTime(t *testing.T, value string) time.Time {
	t.Helper()
	parsed, err := time.Parse(time.RFC3339, value)
	require.NoError(t, err)
	return parsed
}
